REPLACE:dutch_parliament = {
	type = country
	potential = {
		modifier:allow_dutch_parliament = yes
		OR = {
			culture = culture:low_franconian
			culture = { merged_culture_group_contains_culture = culture:low_franconian }
		}
	}
	allow = {
		NOT = { has_country_modifier = supreme_authority_modifier }
		OR = {
			parliament_type = parliament_type:dutch_parliament
			AND = {
				"estate_power(estate_type:burghers_estate)" >= 0.2
				"estate(estate_type:burghers_estate)" = {
					num_privileges >= 3
				}
			}
		}
	}
	modifier = {
		has_a_parliamentary_system = yes
		
		burghers_estate_can_participate_in_parliament = yes
		clergy_estate_can_participate_in_parliament = yes
		global_estate_target_satisfaction = small_permanent_target_satisfaction
		
		nobles_estate_agenda_impact = 0.10
		clergy_estate_agenda_impact = 0.10
		peasants_estate_agenda_impact = 0.10
		burghers_estate_agenda_impact = 0.20
		
		parliament_request_issue_support_needed = -0.10
	}
}
