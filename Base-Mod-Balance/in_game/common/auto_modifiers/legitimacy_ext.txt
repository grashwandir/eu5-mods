# political influence modifier: 0 at 50 legitimacy, full value at 100 (pattern of vanilla high_righteousness)

	potential_trigger = {
		uses_government_power = legitimacy
	}
	limit = {
		legitimacy > 50
	}
	scales_with = {
		value = legitimacy
		subtract = 50
		multiply = 0.02
	}

	monthly_political_influence = 0.25
}
