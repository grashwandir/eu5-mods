protestant_mendicant_order_type = {
	color = { 120 85 50 }
	graphical_cultures = { monastic_gfx }
	location_overlay_loc_key = ORDER_HOLDING
	can_field_levies = no
	default_calling = protestant_almoners_calling
	valid_callings = {
		protestant_preachers_calling
		protestant_almoners_calling
		protestant_confessors_calling
	}
	founding_name_pattern = ORDER_OF_LOCATION_NAME_PATTERN
	founding_cost_gold = 25
	holding_goods_demand = mendicant_order_holding_demand
	default_founding_country_allow = {
		religion = religion:lutheran
	}
	default_founding_location_allow = {
		owner ?= scope:actor
		is_city = yes
	}

	on_leader_death = {
		scope:religious_order = {
			random_pop_in_religious_order_holding_location = {
				age = { 30 50 }
				female = no
				valid_pop_trigger = {
					religion = scope:religious_order.religion
				}
			}
			#### CHANGES ####
			# The spawned head has no country (debug flag): move him to the owner of the location he spawned in
			leader ?= {
				location ?= {
					owner ?= { save_scope_as = lutheran_balance_order_head_country }
				}
				if = {
					limit = { exists = scope:lutheran_balance_order_head_country }
					move_country = scope:lutheran_balance_order_head_country
				}
			}
			#################
		}
	}
}
