######################################################################
# Reverse exclusivity: vanilla market/staple rights also block italy_mercanzia_rights.
# Each block is the vanilla 1.4.x 00_traditions.txt definition + one has_town_rights line.
# A trigger block cannot be INJECTed into an object that already has one (wiki Mod_compatibility),
# so REPLACE is the only way; MEIOU-and-Taxes does the same for its town rights.
# Re-check these copies against vanilla after each game update.
######################################################################

	allow = {
